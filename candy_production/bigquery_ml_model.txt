-- STEP 1: DATA CHECKING

SELECT  
  CAST(observation_date AS DATE) AS date_new,
  IPG3113N
FROM `project-chauhongyennhi-2025.DA04.candy_cleaned`
ORDER BY date_new
;

-- STEP 2: BUILD ARIMA_PLUS MODEL

CREATE OR REPLACE MODEL `project-chauhongyennhi-2025.DA04.candy_production_arima_model`
OPTIONS (
  MODEL_TYPE = 'ARIMA_PLUS',
  TIME_SERIES_TIMESTAMP_COL = 'date_new',
  TIME_SERIES_DATA_COL = 'IPG3113N'
)
AS
SELECT
  CAST(observation_date AS DATE) AS date_new,
  IPG3113N
FROM `project-chauhongyennhi-2025.DA04.candy_cleaned`
ORDER BY date_new
;

-- STEP 3: EVALUATION

SELECT *
FROM ML.ARIMA_EVALUATE(MODEL `project-chauhongyennhi-2025.DA04.candy_production_arima_model`)
;

-- STEP 4: COEFFICIENTS

SELECT *
FROM ML.ARIMA_COEFFICIENTS(MODEL `project-chauhongyennhi-2025.DA04.candy_production_arima_model`)
;

-- BƯỚC 5: FORECAST PRODUCTION FOR THE NEXT 24 MONTHS (KHOẢNG TIN CẬY 95% = 0.95)
-- (HORIZON = 24)

SELECT *
FROM ML.FORECAST(MODEL `project-chauhongyennhi-2025.DA04.candy_production_arima_model`,
                 STRUCT(24 AS HORIZON, 0.95 AS CONFIDENCE_LEVEL)
)
;

-- STEP 6: EXPLAIN FORECAST

SELECT *
FROM ML.EXPLAIN_FORECAST(MODEL `project-chauhongyennhi-2025.DA04.candy_production_arima_model`,
                         STRUCT(24 AS HORIZON, 0.95 AS CONFIDENCE_LEVEL)
)
